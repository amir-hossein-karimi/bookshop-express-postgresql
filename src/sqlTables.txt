authors table

CREATE TABLE authors (
  id      INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  name    TEXT NOT NULL,
  country TEXT
);


books table

CREATE TABLE books (
  id        INTEGER GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  title     TEXT NOT NULL,
  price     NUMERIC(8,2) NOT NULL CHECK (price >= 0),
  author_id INTEGER NOT NULL REFERENCES authors(id)
);
